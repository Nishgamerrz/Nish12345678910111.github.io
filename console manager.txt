#!/data/data/com.termux/files/usr/bin/bash
# Console Manager — complete Termux installer
# Run with: bash "console manager.txt"
# No Python packages, curl, wget, or frontend CDNs are required by this file.

set -euo pipefail
umask 077

if [ -z "${PREFIX:-}" ]; then
    echo "ERROR: PREFIX is not set. Run this installer inside Termux."
    exit 1
fi

DIR="$HOME/.ac-consoles"
PREFERENCES_FILE="$DIR/preferences.json"
INSTALL_LOG="$DIR/install.log"
mkdir -p "$DIR" "$DIR/output" "$DIR/history" "$HOME/.screen"
chmod 700 "$HOME/.screen" "$DIR" "$DIR/output" "$DIR/history"
: > "$INSTALL_LOG"

THEME_IDS=(purple midnight ocean emerald sunset cyber mono contrast)
THEME_NAMES=(Purple Midnight Ocean Emerald Sunset Cyberpunk Monochrome "High Contrast")
ansi_clear() { printf "\033[H\033[2J"; }

valid_theme() {
    local wanted="$1" id
    for id in "${THEME_IDS[@]}"; do [ "$wanted" = "$id" ] && return 0; done
    return 1
}
theme_name() {
    local wanted="$1" i
    for i in "${!THEME_IDS[@]}"; do
        [ "${THEME_IDS[$i]}" = "$wanted" ] && { printf "%s" "${THEME_NAMES[$i]}"; return; }
    done
    printf "Purple"
}
theme_description() {
    case "$1" in
        midnight) printf "Balanced blue on a deep midnight background" ;;
        ocean) printf "Cool cyan accents inspired by deep water" ;;
        purple) printf "Rich violet accents across a deep purple interface" ;;
        emerald) printf "Fresh green accents on dark forest tones" ;;
        sunset) printf "Warm orange and coral on a dark plum background" ;;
        cyber) printf "Electric pink and cyan for a neon terminal" ;;
        mono) printf "Minimal neutral greys with crisp white highlights" ;;
        contrast) printf "Maximum-readability black, white and yellow" ;;
    esac
}
theme_ansi() {
    case "$1" in
        midnight) printf "\033[38;5;75m" ;; ocean) printf "\033[38;5;45m" ;;
        purple) printf "\033[38;5;141m" ;; emerald) printf "\033[38;5;42m" ;;
        sunset) printf "\033[38;5;209m" ;; cyber) printf "\033[38;5;201m" ;;
        mono) printf "\033[38;5;250m" ;; contrast) printf "\033[38;5;226m" ;;
    esac
}
load_saved_theme() {
    local saved=""
    if [ -f "$PREFERENCES_FILE" ]; then
        saved="$(sed -n 's/.*"theme"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p' "$PREFERENCES_FILE" | head -n 1)"
    fi
    if [ -n "$saved" ] && valid_theme "$saved"; then printf "%s" "$saved"; else printf "purple"; fi
}
save_preferences() {
    printf '{\n  "theme": "%s"\n}\n' "$THEME_ID" > "$PREFERENCES_FILE.tmp"
    mv "$PREFERENCES_FILE.tmp" "$PREFERENCES_FILE"
    chmod 600 "$PREFERENCES_FILE"
}
THEME_ID="$(load_saved_theme)"

# A piped installer can still use the phone's terminal through /dev/tty. In a
# truly non-interactive environment it keeps the saved style instead of
# resetting it or blocking forever waiting for input.
INSTALLER_INPUT_FD=0
INSTALLER_INTERACTIVE=0
if [ -t 0 ]; then
    INSTALLER_INTERACTIVE=1
elif (exec 3</dev/tty) 2>/dev/null; then
    # Probe in a subshell first: a failed exec redirection must not terminate
    # a non-interactive reinstall on a device with no controlling terminal.
    exec 3</dev/tty
    INSTALLER_INPUT_FD=3
    INSTALLER_INTERACTIVE=1
fi

installer_draw() {
    local selected="$1" i marker color name
    ansi_clear
    printf "\033[1;35m╔══════════════════════════════════════════════════╗\033[0m\n"
    printf "\033[1;35m║\033[0m  \033[1m%-46s\033[0m\033[1;35m║\033[0m\n" "Console Manager Setup"
    printf "\033[1;35m║\033[0m  %-46s\033[1;35m║\033[0m\n" "Choose a persistent style"
    printf "\033[1;35m╠══════════════════════════════════════════════════╣\033[0m\n"
    for i in "${!THEME_IDS[@]}"; do
        marker=" "
        [ "$i" -eq "$selected" ] && marker=">"
        color="$(theme_ansi "${THEME_IDS[$i]}")"
        name="$(printf "%d. %-12s" "$((i + 1))" "${THEME_NAMES[$i]}")"
        if [ "$i" -eq "$selected" ]; then printf "  \033[7m%s %b%-18s\033[0m" "$marker" "$color" "$name";
        else printf "   %s %b%-18s\033[0m" "$marker" "$color" "$name"; fi
        if [ $((i % 2)) -eq 1 ]; then printf "\n"; else printf "  "; fi
    done
    printf "\033[1;35m╠══════════════════════════════════════════════════╣\033[0m\n"
    printf "  Selected: \033[1m%s\033[0m\n" "$(theme_name "${THEME_IDS[$selected]}")"
    printf "  %s\n" "$(theme_description "${THEME_IDS[$selected]}")"
    printf "  Saved preference: %s\n" "$PREFERENCES_FILE"
    printf "\033[1;35m╠══════════════════════════════════════════════════╣\033[0m\n"
    printf "  [U] Up  [D] Down  [L] Left  [R] Right\n"
    printf "  [1-8] Jump  [Enter] Save + Install  [Q] Quit\n"
    printf "\033[1;35m╚══════════════════════════════════════════════════╝\033[0m\n"
}

select_installer_theme() {
    local selected=0 i key=""
    for i in "${!THEME_IDS[@]}"; do [ "${THEME_IDS[$i]}" = "$THEME_ID" ] && selected="$i"; done
    if [ "$INSTALLER_INTERACTIVE" -eq 0 ]; then
        save_preferences
        printf "Console Manager: using saved %s style.\n" "$(theme_name "$THEME_ID")"
        return
    fi
    while true; do
        installer_draw "$selected"
        IFS= read -rsn1 -u "$INSTALLER_INPUT_FD" key || { save_preferences; return; }
        case "$key" in
            u|U) [ "$selected" -ge 2 ] && selected=$((selected - 2)) ;;
            d|D) [ "$selected" -lt 6 ] && selected=$((selected + 2)) ;;
            l|L) [ $((selected % 2)) -eq 1 ] && selected=$((selected - 1)) ;;
            r|R) [ $((selected % 2)) -eq 0 ] && selected=$((selected + 1)) ;;
            [1-8]) selected=$((key - 1)) ;;
            ""|$'\n'|$'\r')
                THEME_ID="${THEME_IDS[$selected]}"
                save_preferences
                return
                ;;
            q|Q) ansi_clear; echo "Console Manager installation cancelled. Your saved style is unchanged."; exit 0 ;;
        esac
    done
}

dependency_screen() {
    local label="$1" step="$2" total="$3" percent filled empty
    percent=$((step * 100 / total))
    filled=$((percent * 30 / 100))
    empty=$((30 - filled))
    ansi_clear
    printf "\033[1;35m╔══════════════════════════════════════════════════╗\033[0m\n"
    printf "\033[1;35m║\033[0m  \033[1m%-46s\033[0m\033[1;35m║\033[0m\n" "Console Manager Setup"
    printf "\033[1;35m║\033[0m  %-46s\033[1;35m║\033[0m\n" "Installing dependencies"
    printf "\033[1;35m╚══════════════════════════════════════════════════╝\033[0m\n\n"
    printf "  Style: %b%s\033[0m (saved)\n" "$(theme_ansi "$THEME_ID")" "$(theme_name "$THEME_ID")"
    printf "  Step %d/%d\n\n" "$step" "$total"
    printf "  ["; printf "%${filled}s" | tr ' ' '#'; printf "%${empty}s" | tr ' ' '-'; printf "] %3d%%\n\n" "$percent"
    printf "  %s\n" "$label"
}
run_dependency_step() {
    local label="$1" step="$2" total="$3" pid status i=0
    shift 3
    dependency_screen "$label" "$step" "$total"
    printf '\n$ %s\n' "$*" >> "$INSTALL_LOG"
    "$@" >> "$INSTALL_LOG" 2>&1 &
    pid=$!
    local frames=('|' '/' '-' '\')
    while kill -0 "$pid" 2>/dev/null; do
        printf "\r\033[2K  %s %s" "${frames[$((i % 4))]}" "$label"
        i=$((i + 1))
        sleep 0.12
    done
    if wait "$pid"; then status=0; else status=$?; fi
    if [ "$status" -ne 0 ]; then
        printf "\r\033[2K  ERROR: %s failed (%d).\n\n" "$label" "$status"
        tail -n 14 "$INSTALL_LOG"
        exit "$status"
    fi
    printf "\r\033[2K  ✓ %s\n" "$label"
    sleep 0.12
}
install_dependencies() {
    local repair_curl=0 total=2 step=1
    if command -v curl >/dev/null 2>&1 && ! curl --version >/dev/null 2>&1; then repair_curl=1; total=3; fi
    : > "$INSTALL_LOG"
    if [ "${AC_SKIP_INSTALL:-0}" = "1" ]; then
        run_dependency_step "Checking dependencies (install skipped by test mode)" 1 1 true
        return
    fi
    command -v pkg >/dev/null 2>&1 || { echo "ERROR: pkg was not found. Run this installer inside Termux."; exit 1; }
    run_dependency_step "Refreshing Termux package index" "$step" "$total" pkg update
    step=$((step + 1))
    if [ "$repair_curl" -eq 1 ]; then
        run_dependency_step "Repairing curl and Termux libraries" "$step" "$total" pkg upgrade -y
        step=$((step + 1))
    fi
    run_dependency_step "Installing Python and GNU Screen" "$step" "$total" pkg install -y python screen
}

# 1) Style selection happens first. 2) Packages are installed afterwards.
select_installer_theme
install_dependencies
command -v python3 >/dev/null 2>&1 || { echo "ERROR: python3 is required."; exit 1; }
command -v screen >/dev/null 2>&1 || { echo "ERROR: screen is required."; exit 1; }

# Stop only the old Web UI process. Existing screen consoles are left alone.
if [ -f "$DIR/webui.pid" ]; then

if [ -f "$DIR/tailscale/tailscaled.pid" ]; then

        old_ts_cmd=""

            *tailscaled*"$DIR/tailscale/tailscaled.state"*)
                kill "$old_ts_pid" 2>/dev/null || true
                sleep 0.2
                kill -0 "$old_ts_pid" 2>/dev/null && kill -9 "$old_ts_pid" 2>/dev/null || true
                ;;
        esac
    fi
fi
rm -f "$DIR/access-key.txt" "$DIR/https-url.txt"
rm -f "$DIR/bin/tailscale" "$DIR/bin/tailscaled" "$DIR/bin/tailscale.tmp" "$DIR/bin/tailscaled.tmp"
rm -rf "$DIR/tailscale"
for old_ts_archive in "$DIR/downloads"/tailscale_*.tgz; do

def lan_ipv4():
    """Return the phone's local-network IPv4 address without sending traffic."""
    candidates = []
    override = os.environ.get("AC_LAN_IP", "").strip()
    if override:
        candidates.append(override)
    try:
        # UDP connect only selects a route; it does not send a packet.
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock:
            sock.connect(("8.8.8.8", 80))
            candidates.append(sock.getsockname()[0])
    except OSError:
        pass
    try:
        for info in socket.getaddrinfo(socket.gethostname(), 0, socket.AF_INET, socket.SOCK_DGRAM):
            candidates.append(info[4][0])
    except OSError:
        pass
    seen = set()
    for candidate in candidates:
        if candidate in seen:
            continue
        seen.add(candidate)
        try:
            address = ipaddress.IPv4Address(candidate)
        except ipaddress.AddressValueError:
            continue
        if (address.is_private or address.is_link_local) and not address.is_loopback and not address.is_unspecified:
            return str(address)
    return ""

class ApiError(Exception):
    def __init__(self, message, status=400):
        super().__init__(message)
        self.message = message
        self.status = status


def valid_host_header(value):
    # Allow localhost and literal private-network IPs, but reject foreign hostnames
    # to reduce DNS-rebinding exposure while sharing on the same Wi-Fi.
    host = (value or "").strip()
    if host.startswith("[") and "]" in host:
        host = host[1:host.index("]")]
    elif host.count(":") == 1:
        host = host.split(":", 1)[0]
    host = host.lower().rstrip(".")
    allowed = {"localhost", "127.0.0.1", "::1"}
    if WEB_HOST not in ("", "0.0.0.0", "::"):
        allowed.add(WEB_HOST.lower())
    if host in allowed:
        return True
    try:
        address = ipaddress.ip_address(host)
    except ValueError:
        return False
    return address.is_loopback or address.is_private or address.is_link_local


def load_preferences():
    try:
        data = json.loads(PREFERENCES_FILE.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        data = {}
    theme = data.get("theme") if isinstance(data, dict) else None
    return {"theme": theme if theme in THEME_IDS else DEFAULT_THEME}


def save_preferences(theme):
    if theme not in THEME_IDS:
        raise ApiError("Unknown style. Choose one of the available themes.")
    with PREFERENCES_LOCK:
        temporary = PREFERENCES_FILE.with_suffix(".json.tmp")
        temporary.write_text(json.dumps({"theme": theme}, indent=2) + "\n", encoding="utf-8")
        os.replace(temporary, PREFERENCES_FILE)
    return {"theme": theme}


def validate_name(name):
    if not isinstance(name, str) or not NAME_RE.fullmatch(name):
        raise ApiError("Use 1–32 letters, numbers, underscores or hyphens; start with a letter or number.")
    return name


def session_name(name):
    return SESSION_PREFIX + validate_name(name)


def run_screen(args, timeout=6):
    return subprocess.run(
        ["screen", *args],
        env=SCREEN_ENV,
        stdin=subprocess.DEVNULL,
        capture_output=True,
        text=True,
        errors="replace",
        timeout=timeout,
    )


def list_consoles():
    result = run_screen(["-ls"])
    consoles = []
    seen = set()
    for line in result.stdout.splitlines():
        match = SCREEN_LINE_RE.match(line)
        if not match:
            continue
        pid, name = match.groups()
        if name in seen:
            continue
        seen.add(name)
        consoles.append({
            "name": name,
            "session": SESSION_PREFIX + name,
            "pid": int(pid),
            "attached": "(Attached)" in line and "(Detached)" not in line,
            "dead": "dead" in line.lower(),
        })
    consoles.sort(key=lambda item: item["name"].lower())
    return consoles


def find_console(name):
    return next((item for item in list_consoles() if item["name"] == name), None)


def console_exists(name):
    return find_console(name) is not None


def remove_dead_socket(console):
    """Last resort for a confirmed Remote/dead entry only.

    Never unlink a live screen socket. The exact pid + full session name makes
    this unable to touch another console, and screen -wipe is always tried first.
    """
    if not console or not console.get("dead"):
        return
    socket_path = SCREEN_DIR / f"{console['pid']}.{console['session']}"
    try:
        socket_path.unlink()
    except FileNotFoundError:
        pass
    except OSError:
        # The final existence check in delete_console reports a friendly error
        # if even the operating system cannot remove the stale entry.
        pass


def ensure_screenrc():
    if not SCREENRC.exists():
        SCREENRC.write_text(DEFAULT_SCREENRC, encoding="utf-8")


def create_console(name):
    session = session_name(name)
    if console_exists(name):
        raise ApiError("That console already exists.", 409)
    ensure_screenrc()
    bash = shutil.which("bash") or "/data/data/com.termux/files/usr/bin/bash"
    history_file = HISTORY_DIR / (name + ".bash_history")
    env = SCREEN_ENV.copy()
    env["HISTFILE"] = str(history_file)
    env["HISTSIZE"] = "10000"
    env["HISTFILESIZE"] = "20000"
    env["AC_CONSOLE_NAME"] = name
    result = subprocess.run(
        ["screen", "-c", str(SCREENRC), "-dmS", session, bash],
        env=env,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
        timeout=6,
    )
    if result.returncode != 0:
        raise ApiError("screen could not create the console; check the Web UI log.", 500)
    for _ in range(20):
        if console_exists(name):
            return
        time.sleep(0.1)
    raise ApiError("The console was started but did not appear in screen -ls.", 500)


def delete_console(name):
    session = session_name(name)
    if find_console(name) is None:
        return  # idempotent: deleting an already-gone console is still success

    # A live/attached console closes here. Remote or dead sessions cannot
    # receive -X quit, so never show screen's raw error: continue to wipe them.
    try:
        run_screen(["-S", session, "-X", "quit"])
    except (OSError, subprocess.TimeoutExpired):
        pass
    time.sleep(0.15)

    # screen -wipe [match] removes destroyed/unreachable sessions. Target only
    # this console; do NOT wipe unrelated dead screens belonging to the user.
    if find_console(name) is not None:
        try:
            run_screen(["-wipe", session])
        except (OSError, subprocess.TimeoutExpired):
            pass
        time.sleep(0.1)

    # Some Termux/screen builds leave a stale socket even after targeted wipe.
    # Remove it only while screen -ls still explicitly marks THIS entry dead.
    current = find_console(name)
    if current is not None and current.get("dead"):
        remove_dead_socket(current)

    # Retry with the unique pid.session matcher used by screen itself.
    current = find_console(name)
    if current is not None and current.get("dead"):
        try:
            run_screen(["-wipe", f"{current['pid']}.{session}"])
        except (OSError, subprocess.TimeoutExpired):
            pass
        current = find_console(name)
        if current is not None and current.get("dead"):
            remove_dead_socket(current)

    if find_console(name) is not None:
        raise ApiError("This console is still active and could not be closed. Open it, run 'exit', then try Delete again.", 500)


def clean_hardcopy(raw):
    text = ANSI_RE.sub("", raw).replace("\x00", "").replace("\r\n", "\n").replace("\r", "\n")
    lines = [line.rstrip() for line in text.split("\n")]
    while lines and not lines[-1]:
        lines.pop()
    text = "\n".join(lines)
    truncated = len(text) > MAX_OUTPUT_CHARS
    if truncated:
        text = "… older output truncated by the Web UI …\n" + text[-MAX_OUTPUT_CHARS:]
    return text, truncated


def read_output(name):
    session = session_name(name)
    fd, path_string = tempfile.mkstemp(prefix=name + "-", suffix=".hardcopy", dir=str(OUTPUT_DIR))
    os.close(fd)
    path = Path(path_string)
    try:
        # screen expects to create the hardcopy itself.
        path.unlink()
        result = run_screen(["-S", session, "-X", "hardcopy", "-h", str(path)])
        if result.returncode != 0 or not path.exists():
            detail = (result.stderr or result.stdout).strip()
            raise ApiError("Console not found or its output is unavailable." + (" " + detail if detail else ""), 404)
        raw = path.read_text(encoding="utf-8", errors="replace")
        text, truncated = clean_hardcopy(raw)
        return {"output": text, "truncated": truncated}
    finally:
        try:
            path.unlink()
        except FileNotFoundError:
            pass


def remember_command(name, command):
    # Commands entered through the Web UI also become completion suggestions.
    # The full output/history still comes from the real screen scrollback.
    path = HISTORY_DIR / (name + ".web_history")
    with HISTORY_LOCK:
        with path.open("a", encoding="utf-8") as handle:
            handle.write(command.replace("\n", " ") + "\n")
        try:
            lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
            if len(lines) > 300:
                path.write_text("\n".join(lines[-200:]) + "\n", encoding="utf-8")
        except OSError:
            pass


def execute_command(name, command):
    session = session_name(name)
    if not isinstance(command, str) or not command.strip():
        raise ApiError("Enter a command first.")
    if len(command) > MAX_COMMAND_CHARS:
        raise ApiError("Command is too long (4000 characters maximum).")
    if "\x00" in command or "\n" in command or "\r" in command:
        raise ApiError("Enter one command line at a time.")
    if not console_exists(name):
        raise ApiError("Console not found.", 404)
    # No shell=True here: the command is pasted into the existing interactive
    # shell, exactly as if it was typed in Termux. This also preserves Termux's
    # own command-not-found / "did you mean" output.
    result = run_screen(["-S", session, "-X", "stuff", command + "\n"])
    if result.returncode != 0:
        raise ApiError("Could not send the command: " + (result.stderr.strip() or "screen error"), 500)
    remember_command(name, command)


def recent_web_history(name):
    path = HISTORY_DIR / (name + ".web_history")
    try:
        lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    except OSError:
        return []
    return list(reversed(lines[-150:]))


def available_commands():
    now = time.monotonic()
    with COMMAND_CACHE_LOCK:
        if now - COMMAND_CACHE["at"] < 60 and COMMAND_CACHE["commands"]:
            return COMMAND_CACHE["commands"]
    commands = {
        "alias", "bg", "break", "cd", "clear", "continue", "declare", "dirs",
        "disown", "echo", "eval", "exec", "exit", "export", "fg", "help",
        "history", "jobs", "kill", "popd", "printf", "pushd", "pwd", "read",
        "readonly", "set", "shift", "source", "test", "times", "trap", "type",
        "typeset", "ulimit", "umask", "unalias", "unset", "wait",
    }
    for folder in os.environ.get("PATH", "").split(os.pathsep):
        if not folder:
            continue
        try:
            with os.scandir(folder) as entries:
                for entry in entries:
                    try:
                        if entry.is_file() and os.access(entry.path, os.X_OK):
                            commands.add(entry.name)
                    except OSError:
                        continue
        except OSError:
            continue
    result = sorted(commands)
    with COMMAND_CACHE_LOCK:
        COMMAND_CACHE["at"] = now
        COMMAND_CACHE["commands"] = result
    return result


def suggest_commands(name, query):
    validate_name(name)
    if not query or len(query) > 80:
        return []
    lowered = query.lower()
    suggestions = []
    seen = set()
    for command in recent_web_history(name) + available_commands():
        if command.lower().startswith(lowered) and command not in seen:
            suggestions.append(command)
            seen.add(command)
        if len(suggestions) >= 12:
            break
    return suggestions


def get_stats():
    memory = {"used": 0, "total": 0, "percent": 0}
    try:
        values = {}
        with open("/proc/meminfo", encoding="utf-8") as handle:
            for line in handle:
                parts = line.split()
                if len(parts) >= 2 and parts[0].endswith(":"):
                    values[parts[0][:-1]] = int(parts[1])
        total = values.get("MemTotal", 0) // 1024
        available = values.get("MemAvailable")
        if available is None:
            available = values.get("MemFree", 0) + values.get("Buffers", 0) + values.get("Cached", 0)
        free = available // 1024
        used = max(0, total - free)
        memory = {"used": used, "total": total, "percent": round(used * 100 / total) if total else 0}
    except (OSError, ValueError):
        pass
    try:
        with open("/proc/loadavg", encoding="utf-8") as handle:
            load = " ".join(handle.read().split()[:3])
    except OSError:
        load = "N/A"
    disk = {"used": "", "total": "", "percent": ""}
    try:
        usage = shutil.disk_usage(str(Path.home()))
        disk = {
            "used": format_bytes(usage.used),
            "total": format_bytes(usage.total),
            "percent": str(round(usage.used * 100 / usage.total)) + "%" if usage.total else "0%",
        }
    except OSError:
        pass
    return {"load": load, "memory": memory, "disk": disk}


def format_bytes(value):
    value = float(value)
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if value < 1024 or unit == "TB":
            return f"{value:.1f} {unit}" if unit != "B" else f"{int(value)} B"
        value /= 1024


class Handler(http.server.BaseHTTPRequestHandler):
    server_version = "ConsoleManager/3.0"
    sys_version = ""

    def request_allowed(self):
        if not valid_host_header(self.headers.get("Host", "")):
            self.send_json({"ok": False, "error": "This hostname is not allowed."}, 403)
            return False
        return True

    # Never write HTTP access logs to the terminal running the ac dashboard.
    def log_message(self, *args):
        return

    def common_headers(self):
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("Referrer-Policy", "no-referrer")
        self.send_header("Cache-Control", "no-store")
        self.send_header("Content-Security-Policy", "default-src 'self'; script-src 'unsafe-inline'; style-src 'unsafe-inline'; connect-src 'self'; img-src 'self' data:; base-uri 'none'; frame-ancestors 'none'")

    def send_json(self, data, code=200):
        body = json.dumps(data, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.common_headers()
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def send_html(self):
        page = HTML.replace("__INITIAL_THEME__", load_preferences()["theme"])
        body = page.encode("utf-8")
        self.send_response(200)
        self.common_headers()
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def read_body(self):
        # JSON requests trigger a browser CORS preflight cross-origin. Rejecting
        # simple form/text POSTs helps protect the password-free LAN dashboard.
        content_type = self.headers.get("Content-Type", "").split(";", 1)[0].strip().lower()
        if content_type != "application/json":
            raise ApiError("Content-Type must be application/json.", 415)
        try:
            length = int(self.headers.get("Content-Length", "0"))
        except ValueError:
            raise ApiError("Invalid Content-Length.")
        if length < 0 or length > MAX_BODY_BYTES:
            raise ApiError("Request body is too large.", 413)
        if not length:
            return {}
        try:
            data = json.loads(self.rfile.read(length))
        except (json.JSONDecodeError, UnicodeDecodeError):
            raise ApiError("Invalid JSON request.")
        if not isinstance(data, dict):
            raise ApiError("JSON body must be an object.")
        return data

    def handle_exception(self, error):
        if isinstance(error, ApiError):
            self.send_json({"ok": False, "error": error.message}, error.status)
        elif isinstance(error, subprocess.TimeoutExpired):
            self.send_json({"ok": False, "error": "A screen operation timed out. Try again."}, 504)
        elif isinstance(error, (BrokenPipeError, ConnectionResetError)):
            return
        else:
            # This goes to webui.log because webui_start redirects both stdout
            # and stderr there. It can never draw over the interactive dashboard.
            traceback.print_exc()
            self.send_json({"ok": False, "error": "Internal Web UI error; see ~/.ac-consoles/webui.log."}, 500)

    def do_GET(self):
        if not self.request_allowed():
            return
        try:
            parsed = urllib.parse.urlsplit(self.path)
            query = urllib.parse.parse_qs(parsed.query)
            if parsed.path in ("/", "/index.html"):
                self.send_html()
            elif parsed.path == "/api/state":
                self.send_json({"ok": True, "consoles": list_consoles(), "stats": get_stats(), "preferences": load_preferences(), "themes": WEB_THEMES})
            elif parsed.path == "/api/output":
                name = validate_name(query.get("name", [""])[0])
                data = read_output(name)
                self.send_json({"ok": True, "name": name, **data})
            elif parsed.path == "/api/suggest":
                name = query.get("name", [""])[0]
                q = query.get("q", [""])[0]
                self.send_json({"ok": True, "suggestions": suggest_commands(name, q)})
            else:
                self.send_json({"ok": False, "error": "Not found."}, 404)
        except Exception as error:
            self.handle_exception(error)

    def do_POST(self):
        if not self.request_allowed():
            return
        try:
            path = urllib.parse.urlsplit(self.path).path
            body = self.read_body()
            if path == "/api/create":
                name = validate_name(body.get("name", ""))
                create_console(name)
                self.send_json({"ok": True, "name": name})
            elif path == "/api/delete":
                name = validate_name(body.get("name", ""))
                delete_console(name)
                self.send_json({"ok": True})
            elif path == "/api/preferences":
                preferences = save_preferences(body.get("theme", ""))
                self.send_json({"ok": True, "preferences": preferences})
            elif path == "/api/execute":
                name = validate_name(body.get("name", ""))
                execute_command(name, body.get("command", ""))
                self.send_json({"ok": True})
            else:
                self.send_json({"ok": False, "error": "Not found."}, 404)
        except Exception as error:
            self.handle_exception(error)

webui_start() {
    if webui_running; then
        echo "Web UI is already running."
        WEBUI_IP_TRIED=0
        show_webui_urls
        return 0
    fi
    rm -f "$WEBUI_PID_FILE"
    [ -f "$WEBUI_PY" ] || { echo "ERROR: Web UI files are missing; rerun 'console manager.txt'."; return 1; }

    # Fully detach stdin/stdout/stderr. This is the fix for HTTP/Python log
    # lines appearing over the ac dashboard: the daemon has no terminal to
    # write to, and BaseHTTPRequestHandler.log_message is disabled as well.
    if command -v setsid >/dev/null 2>&1; then
        nohup setsid env AC_CONSOLE_DIR="$DIR" SCREENDIR="$SCREENDIR" \
            AC_WEB_HOST="$WEB_HOST" AC_WEB_PORT="$WEB_PORT" \
            python3 "$WEBUI_PY" </dev/null >>"$WEBUI_LOG" 2>&1 &
    else
        nohup env AC_CONSOLE_DIR="$DIR" SCREENDIR="$SCREENDIR" \
            AC_WEB_HOST="$WEB_HOST" AC_WEB_PORT="$WEB_PORT" \
            python3 "$WEBUI_PY" </dev/null >>"$WEBUI_LOG" 2>&1 &
    fi
    local pid=$!
    echo "$pid" > "$WEBUI_PID_FILE"
    sleep 0.8
    if webui_running; then
        echo "Web UI started."
        WEBUI_IP_TRIED=0
        show_webui_urls
        echo "Logs stay in $WEBUI_LOG (not on the dashboard)."
    else
        rm -f "$WEBUI_PID_FILE"
        echo "ERROR: Web UI failed to start. Check $WEBUI_LOG"
        return 1
    fi
}

webui_stop() {
    local pid
    if webui_running; then
        pid="$(webui_pid)"
        kill "$pid" 2>/dev/null
        sleep 0.3
        kill -0 "$pid" 2>/dev/null && kill -9 "$pid" 2>/dev/null
        rm -f "$WEBUI_PID_FILE"
        echo "Web UI stopped."
    else
        rm -f "$WEBUI_PID_FILE"
        echo "Web UI is not running."
    fi
    WEBUI_LAN_IP=""
    WEBUI_IP_TRIED=0
}

console_is_dead() { console_line "$1" | grep -qi 'dead'; }
remove_dead_socket() {
    local name="$1" pid=""
    console_is_dead "$name" || return 0
    pid="$(console_pid "$name")"
    if [ -n "$pid" ]; then
        rm -f -- "$SCREENDIR/$pid.${SESSION_PREFIX}${name}" 2>/dev/null || true
    fi
}
valid_name() { [[ "$1" =~ ^[A-Za-z0-9][A-Za-z0-9_-]{0,31}$ ]]; }

create_console() {
    local name="$1"
    [ -n "$name" ] || { echo "Usage: ac create <name>"; return 1; }
    valid_name "$name" || { echo "Use 1-32 letters, numbers, _ or -; start with a letter or number."; return 1; }
    if exists "$name"; then echo "Console '$name' already exists."; return 0; fi
    screen -c "$RC" -dmS "${SESSION_PREFIX}${name}" bash
    sleep 0.4
    exists "$name" && echo "Console '$name' created." || { echo "ERROR: failed to create console."; return 1; }
}

open_console() {
    local name="$1"
    [ -n "$name" ] || { echo "Usage: ac open <name>"; return 1; }
    exists "$name" || { echo "Console '$name' not found."; return 1; }
    screen -S "${SESSION_PREFIX}${name}" -X bindkey "qq" detach 2>/dev/null
    screen -r "${SESSION_PREFIX}${name}"
}

delete_console() {
    local name="$1" pid=""
    [ -n "$name" ] || { echo "Usage: ac delete <name>"; return 1; }
    if ! exists "$name"; then
        echo "Console '$name' is already gone."
        return 0
    fi

    # Live consoles close with quit. Remote/dead consoles reject it, so hide
    # that expected screen error and continue through the stale-session cleanup.
    screen -S "${SESSION_PREFIX}${name}" -X quit >/dev/null 2>&1 || true
    sleep 0.15

    # Target only this session; never wipe unrelated screens.
    if exists "$name"; then
        screen -wipe "${SESSION_PREFIX}${name}" >/dev/null 2>&1 || true
        sleep 0.1
    fi

    # Fallback for Termux/screen builds that leave the dead socket behind.
    if exists "$name" && console_is_dead "$name"; then
        remove_dead_socket "$name"
    fi

    # Retry using screen's unique pid.session matcher, then clean the exact
    # socket again if it is still explicitly marked Remote or dead.
    if exists "$name" && console_is_dead "$name"; then
        pid="$(console_pid "$name")"
        if [ -n "$pid" ]; then
            screen -wipe "$pid.${SESSION_PREFIX}${name}" >/dev/null 2>&1 || true
        fi
        if exists "$name" && console_is_dead "$name"; then
            remove_dead_socket "$name"
        fi
    fi

    if exists "$name"; then
        echo "ERROR: Console '$name' is still active. Open it, run 'exit', then delete it again."
        return 1
    fi
    echo "Console '$name' deleted."
}

get_stats() {
    local mem_used mem_total mem_pct disk_pct load web
    mem_used=$(free -m 2>/dev/null | awk '/Mem:/{print $3}')
    mem_total=$(free -m 2>/dev/null | awk '/Mem:/{print $2}')
    if [ -n "$mem_used" ] && [ -n "$mem_total" ] && [ "$mem_total" -gt 0 ] 2>/dev/null; then
        mem_pct=$(awk "BEGIN{printf \"%d\", $mem_used*100/$mem_total}")
    else mem_pct="?"; fi
    disk_pct=$(df -h "$HOME" 2>/dev/null | awk 'NR==2{print $5}')
    load=$(awk '{print $1}' /proc/loadavg 2>/dev/null || echo "?")
    web=$(webui_running && printf "ON" || printf "OFF")
    printf "RAM:%s%% Disk:%s Web:%s L:%s" "$mem_pct" "${disk_pct:-?}" "$web" "$load"
}

STATS_ROW=5
CONSOLES=()
SELECTED_INDEX=0

draw_full() {
    local stats count i name label status marker
    stats=$(get_stats)
    count=${#CONSOLES[@]}
    ansi_clear
    printf "%b╔════════════════════════════════════════╗%b\n" "$C_BORDER" "$C_RESET"
    printf "%b║%b  %b%-38s%b%b║%b\n" "$C_BORDER" "$C_RESET" "$C_ACCENT$C_BOLD" "Console Manager" "$C_RESET" "$C_BORDER" "$C_RESET"
    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_MUTED" "Style: $(theme_name "$THEME_ID") | $count console(s)" "$C_BORDER" "$C_RESET"
    printf "%b╠════════════════════════════════════════╣%b\n" "$C_BORDER" "$C_RESET"
    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_TEXT" "$stats" "$C_BORDER" "$C_RESET"
    printf "%b╠════════════════════════════════════════╣%b\n" "$C_BORDER" "$C_RESET"
    if [ "$count" -eq 0 ]; then
        printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_MUTED" "  No consoles — press C to create" "$C_BORDER" "$C_RESET"
    else
        for i in "${!CONSOLES[@]}"; do
            name="${CONSOLES[$i]}"
            status="live"
            console_is_dead "$name" && status="dead"
            marker=" "
            [ "$i" -eq "$SELECTED_INDEX" ] && marker=">"
            label="$(printf "%s %-25s [%s]" "$marker" "$name" "$status")"
            if [ "$i" -eq "$SELECTED_INDEX" ]; then
                printf "%b║  %b%-38s%b%b║%b\n" "$C_BORDER" "$C_SELECTED" "$label" "$C_RESET" "$C_BORDER" "$C_RESET"
            else
                printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_TEXT" "$label" "$C_BORDER" "$C_RESET"
            fi
        done
    fi
    printf "%b╠════════════════════════════════════════╣%b\n" "$C_BORDER" "$C_RESET"
    if webui_running; then
        refresh_webui_ip
        printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_ACCENT" "Local: http://localhost:$WEB_PORT" "$C_BORDER" "$C_RESET"
        case "$WEB_HOST" in
            127.0.0.1|localhost|::1)
                printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_MUTED" "Wi-Fi: disabled" "$C_BORDER" "$C_RESET" ;;
            *)
                if [ -n "$WEBUI_LAN_IP" ]; then
                    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_ACCENT" "Wi-Fi: http://$WEBUI_LAN_IP:$WEB_PORT" "$C_BORDER" "$C_RESET"
                else
                    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_MUTED" "Wi-Fi: IP not detected" "$C_BORDER" "$C_RESET"
                fi ;;
        esac
        printf "%b╠════════════════════════════════════════╣%b\n" "$C_BORDER" "$C_RESET"
    fi
    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_ACCENT" "[U]Up [D]Down [L]Left [R]Right" "$C_BORDER" "$C_RESET"
    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_TEXT" "[Enter]Open [C]reate [X]Delete" "$C_BORDER" "$C_RESET"
    printf "%b║%b  %-38s%b║%b\n" "$C_BORDER" "$C_TEXT" "[T]Style [N]Web [H]elp [Q]uit" "$C_BORDER" "$C_RESET"
    printf "%b╚════════════════════════════════════════╝%b\n" "$C_BORDER" "$C_RESET"
}

update_stats_inline() {
    local stats
    stats=$(get_stats)
    ansi_save
    ansi_goto "$STATS_ROW"
    printf "\033[2K%b║%b  %-38s%b║%b" "$C_BORDER" "$C_TEXT" "$stats" "$C_BORDER" "$C_RESET"
    ansi_restore
}

dashboard() {
    local prev_names="" prev_stats="" curr_names curr_stats drawn=0 choice n count
    while true; do
        mapfile -t CONSOLES < <(list_names | sort)
        count=${#CONSOLES[@]}
        if [ "$count" -eq 0 ]; then SELECTED_INDEX=0;
        elif [ "$SELECTED_INDEX" -ge "$count" ]; then SELECTED_INDEX=$((count - 1)); fi
        curr_names=$(printf '%s|' "${CONSOLES[@]}")
        curr_stats=$(get_stats)
        if [ "$curr_names" != "$prev_names" ] || [ "$drawn" -eq 0 ]; then
            draw_full
            prev_names="$curr_names"
            prev_stats="$curr_stats"
            drawn=1
        elif [ "$curr_stats" != "$prev_stats" ]; then
            update_stats_inline
            prev_stats="$curr_stats"
        fi
        if read -t 1 -rsn1 choice 2>/dev/null; then
            case "$choice" in
                u|U|l|L)
                    [ "$count" -gt 0 ] && SELECTED_INDEX=$(((SELECTED_INDEX - 1 + count) % count))
                    draw_full ;;
                d|D|r|R)
                    [ "$count" -gt 0 ] && SELECTED_INDEX=$(((SELECTED_INDEX + 1) % count))
                    draw_full ;;
                ""|$'\n'|$'\r')
                    if [ "$count" -gt 0 ]; then open_console "${CONSOLES[$SELECTED_INDEX]}"; drawn=0; fi ;;
                c|C)
                    printf "\n"; read -rp "New console name: " n
                    [ -n "$n" ] && create_console "$n"
                    sleep 0.3; drawn=0 ;;
                x|X)
                    if [ "$count" -gt 0 ]; then
                        printf "\n"; read -rp "Delete '${CONSOLES[$SELECTED_INDEX]}'? [y/N] " n
                        case "$n" in y|Y|yes|YES) delete_console "${CONSOLES[$SELECTED_INDEX]}" ;; esac
                    fi
                    sleep 0.3; drawn=0 ;;
                t|T) cycle_theme; draw_full ;;
                n|N) printf "\n"; webui_toggle; sleep 0.5; drawn=0 ;;
                h|H) ansi_clear; show_help; read -rp "Press Enter to return..."; drawn=0 ;;
                q|Q|0) ansi_clear; exit 0 ;;
            esac
        fi
    done
}

case "${1:-}" in
    list) list_names ;;
    create) create_console "${2:-}" ;;
    open) open_console "${2:-}" ;;
    delete) delete_console "${2:-}" ;;
    theme) set_theme_command "${2:-}" ;;
    webui) webui_toggle ;;
    webui-start) webui_start ;;
    webui-stop) webui_stop ;;
    webui-status)
        webui_status; echo
        if webui_running; then WEBUI_IP_TRIED=0; show_webui_urls; fi ;;
    webui-urls) WEBUI_IP_TRIED=0; show_webui_urls ;;
    help|--help|-h) show_help ;;
    "") dashboard ;;
    *) echo "Unknown command."; show_help; exit 1 ;;
esac
