#!/data/data/com.termux/files/usr/bin/bash
# Aurer Console — fixed Termux installer
# Run with: bash "console manager.txt"
# No Python packages or frontend CDNs are required.

set -euo pipefail

if [ -z "${PREFIX:-}" ]; then
    echo "ERROR: PREFIX is not set. Run this installer inside Termux."
    exit 1
fi

if [ "${AC_SKIP_INSTALL:-0}" != "1" ]; then
    if command -v pkg >/dev/null 2>&1; then
        pkg install -y python screen
    else
        echo "ERROR: pkg was not found. Run this installer inside Termux."
        exit 1
    fi
fi

command -v python3 >/dev/null 2>&1 || { echo "ERROR: python3 is required."; exit 1; }
command -v screen >/dev/null 2>&1 || { echo "ERROR: screen is required."; exit 1; }

DIR="$HOME/.ac-consoles"
mkdir -p "$DIR" "$DIR/output" "$DIR/history" "$HOME/.screen"
chmod 700 "$HOME/.screen" "$DIR" "$DIR/output" "$DIR/history"

# Stop only the old Web UI process. Existing screen consoles are left alone.
if [ -f "$DIR/webui.pid" ]; then
    old_pid="$(cat "$DIR/webui.pid" 2>/dev/null || true)"
    if [[ "$old_pid" =~ ^[0-9]+$ ]] && kill -0 "$old_pid" 2>/dev/null; then
        kill "$old_pid" 2>/dev/null || true
        sleep 0.3
    fi
    rm -f "$DIR/webui.pid"
fi

cat > "$DIR/webui.py" <<'PYEOF'

<script>
'use strict';
const byId = id => document.getElementById(id);
const state = {
  consoles: [], active: null, output: '', outputBusy: false,
  suggestions: [], suggestionIndex: -1, suggestTimer: null, toastTimer: null
};

async function api(path, options = {}) {
  const response = await fetch(path, { cache: 'no-store', ...options });
  let data;
  try { data = await response.json(); }
  catch { throw new Error('The Web UI returned an invalid response.'); }
  if (!response.ok || data.ok === false) throw new Error(data.error || `Request failed (${response.status})`);
  return data;
}
function jsonOptions(body) {
  return { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) };
}
function notify(message, isError = false) {
  const el = byId('toast');
  el.textContent = message;
  el.classList.toggle('error', isError);
  el.classList.add('show');
  clearTimeout(state.toastTimer);
  state.toastTimer = setTimeout(() => el.classList.remove('show'), 2600);
}
function renderStats(stats) {
  byId('loadStat').textContent = stats.load || 'N/A';
  byId('memoryStat').textContent = stats.memory.total ? `${stats.memory.used}/${stats.memory.total} MB · ${stats.memory.percent}%` : 'N/A';
  byId('diskStat').textContent = stats.disk.total ? `${stats.disk.used}/${stats.disk.total} · ${stats.disk.percent}` : 'N/A';
}
function makeButton(label, className, onClick) {
  const button = document.createElement('button');
  button.type = 'button'; button.className = `btn ${className}`; button.textContent = label;
  button.addEventListener('click', onClick);
  return button;
}
function renderConsoles() {
  const list = byId('consoleList');
  list.replaceChildren();
  byId('consoleCount').textContent = String(state.consoles.length);
  if (!state.consoles.length) {
    const empty = document.createElement('div'); empty.className = 'empty';
    const strong = document.createElement('strong'); strong.textContent = 'No consoles running';
    empty.append(strong, document.createTextNode('Create one above to open a real Termux shell.'));
    list.append(empty); return;
  }
  for (const consoleInfo of state.consoles) {
    const card = document.createElement('article'); card.className = 'console-card';
    const glyph = document.createElement('div'); glyph.className = 'console-glyph'; glyph.textContent = '>_';
    const meta = document.createElement('div'); meta.className = 'console-meta';
    const name = document.createElement('div'); name.className = 'console-name'; name.textContent = consoleInfo.name;
    const sub = document.createElement('div'); sub.className = 'console-state';
    sub.textContent = consoleInfo.dead ? 'Remote or dead · Delete will clean it up' : (consoleInfo.attached ? 'Screen attached · output available' : 'Screen detached · output available');
    meta.append(name, sub);
    const actions = document.createElement('div'); actions.className = 'card-actions';
    actions.append(
      makeButton('Open console', 'btn-primary', () => openConsole(consoleInfo.name)),
      makeButton('Delete', 'btn-danger', () => deleteConsole(consoleInfo.name))
    );
    card.append(glyph, meta, actions); list.append(card);
  }
}
async function refreshDashboard(showErrors = false) {
  if (state.active) return;
  try {
    const data = await api('/api/state');
    state.consoles = data.consoles; renderStats(data.stats); renderConsoles();
    byId('dashboardError').hidden = true;
  } catch (error) {
    if (showErrors || !state.consoles.length) {
      const banner = byId('dashboardError'); banner.textContent = error.message; banner.hidden = false;
    }
  }
}
async function createConsole(event) {
  event.preventDefault();
  const input = byId('newName'); const name = input.value.trim();
  if (!name) return;
  try {
    await api('/api/create', jsonOptions({ name }));
    input.value = ''; notify(`Console “${name}” created.`); await refreshDashboard(true);
  } catch (error) { notify(error.message, true); }
}
async function deleteConsole(name) {
  if (!window.confirm(`Delete console “${name}”? Its screen session will be closed.`)) return;
  try {
    await api('/api/delete', jsonOptions({ name }));
    notify(`Console “${name}” deleted.`); await refreshDashboard(true);
  } catch (error) { notify(error.message, true); }
}
async function openConsole(name) {
  state.active = name; state.output = ''; hideSuggestions();
  byId('dashboardView').hidden = true; byId('consoleView').hidden = false;
  byId('activeName').textContent = name;
  byId('terminalOutput').textContent = 'Loading console output…';
  history.replaceState(null, '', `#${encodeURIComponent(name)}`);
  await loadOutput(true);
  byId('commandInput').focus();
}
function closeConsole() {
  state.active = null; state.output = ''; hideSuggestions();
  byId('consoleView').hidden = true; byId('dashboardView').hidden = false;
  history.replaceState(null, '', location.pathname);
  refreshDashboard(true);
}
async function loadOutput(forceScroll = false) {
  const name = state.active;
  if (!name || state.outputBusy) return;
  state.outputBusy = true;
  try {
    const data = await api(`/api/output?name=${encodeURIComponent(name)}`);
    if (state.active !== name) return;
    if (data.output !== state.output || forceScroll) {
      const terminal = byId('terminalOutput');
      const nearBottom = terminal.scrollHeight - terminal.scrollTop - terminal.clientHeight < 90;
      state.output = data.output;
      terminal.textContent = data.output || '(No output in this console yet.)';
      if (forceScroll || nearBottom) terminal.scrollTop = terminal.scrollHeight;
    }
  } catch (error) {
    if (state.active === name) notify(error.message, true);
  } finally { state.outputBusy = false; }
}
async function executeCommand(event) {
  event.preventDefault();
  const input = byId('commandInput'); const command = input.value;
  if (!command.trim() || !state.active) return;
  hideSuggestions();
  const button = byId('runButton'); button.disabled = true;
  try {
    await api('/api/execute', jsonOptions({ name: state.active, command }));
    input.value = '';
    setTimeout(() => loadOutput(), 250);
  } catch (error) { notify(error.message, true); }
  finally { button.disabled = false; input.focus(); }
}
function hideSuggestions() {
  state.suggestions = []; state.suggestionIndex = -1;
  byId('suggestions').hidden = true; byId('suggestions').replaceChildren();
}
function acceptSuggestion(index) {
  if (!state.suggestions.length) return;
  const selected = state.suggestions[Math.max(0, Math.min(index, state.suggestions.length - 1))];
  byId('commandInput').value = selected; hideSuggestions(); byId('commandInput').focus();
}
function renderSuggestions() {
  const box = byId('suggestions'); box.replaceChildren();
  if (!state.suggestions.length) { box.hidden = true; return; }
  const label = document.createElement('div'); label.className = 'suggestion-label'; label.textContent = 'Did you mean… (Tab completes)'; box.append(label);
  state.suggestions.forEach((suggestion, index) => {
    const item = document.createElement('button'); item.type = 'button';
    item.className = `suggestion${index === state.suggestionIndex ? ' active' : ''}`;
    item.textContent = suggestion;
    item.addEventListener('mousedown', event => event.preventDefault());
    item.addEventListener('click', () => acceptSuggestion(index)); box.append(item);
  });
  box.hidden = false;
}
function queueSuggestions() {
  clearTimeout(state.suggestTimer);
  const query = byId('commandInput').value.trim();
  if (!query || query.length > 80 || /\s/.test(query) || !state.active) { hideSuggestions(); return; }
  const consoleName = state.active;
  state.suggestTimer = setTimeout(async () => {
    try {
      const data = await api(`/api/suggest?name=${encodeURIComponent(consoleName)}&q=${encodeURIComponent(query)}`);
      if (state.active !== consoleName || byId('commandInput').value.trim() !== query || document.activeElement !== byId('commandInput')) return;
      state.suggestions = data.suggestions; state.suggestionIndex = -1; renderSuggestions();
    } catch { hideSuggestions(); }
  }, 120);
}
function commandKeys(event) {
  if (event.key === 'Escape') { hideSuggestions(); return; }
  if (!state.suggestions.length) return;
  if (event.key === 'ArrowDown' || event.key === 'ArrowUp') {
    event.preventDefault();
    if (state.suggestionIndex < 0) state.suggestionIndex = event.key === 'ArrowDown' ? 0 : state.suggestions.length - 1;
    else {
      const direction = event.key === 'ArrowDown' ? 1 : -1;
      state.suggestionIndex = (state.suggestionIndex + direction + state.suggestions.length) % state.suggestions.length;
    }
    renderSuggestions();
  } else if (event.key === 'Tab') {
    event.preventDefault(); acceptSuggestion(state.suggestionIndex < 0 ? 0 : state.suggestionIndex);
  } else if (event.key === 'Enter' && state.suggestionIndex >= 0) {
    event.preventDefault(); acceptSuggestion(state.suggestionIndex);

class ApiError(Exception):
    def __init__(self, message, status=400):
        super().__init__(message)
        self.message = message
        self.status = status


def validate_name(name):
    if not isinstance(name, str) or not NAME_RE.fullmatch(name):
        raise ApiError("Use 1–32 letters, numbers, underscores or hyphens; start with a letter or number.")
    return name


def session_name(name):
    return SESSION_PREFIX + validate_name(name)


def run_screen(args, timeout=6):
    return subprocess.run(
        ["screen", *args],
        env=SCREEN_ENV,
        stdin=subprocess.DEVNULL,
        capture_output=True,
        text=True,
        errors="replace",
        timeout=timeout,
    )


def list_consoles():
    result = run_screen(["-ls"])
    consoles = []
    seen = set()
    for line in result.stdout.splitlines():
        match = SCREEN_LINE_RE.match(line)
        if not match:
            continue
        pid, name = match.groups()
        if name in seen:
            continue
        seen.add(name)
        consoles.append({
            "name": name,
            "session": SESSION_PREFIX + name,
            "pid": int(pid),
            "attached": "(Attached)" in line and "(Detached)" not in line,
            "dead": "dead" in line.lower(),
        })
    consoles.sort(key=lambda item: item["name"].lower())
    return consoles


def find_console(name):
    return next((item for item in list_consoles() if item["name"] == name), None)


def console_exists(name):
    return find_console(name) is not None


def remove_dead_socket(console):
    """Last resort for a confirmed Remote/dead entry only.

    Never unlink a live screen socket. The exact pid + full session name makes
    this unable to touch another console, and screen -wipe is always tried first.
    """
    if not console or not console.get("dead"):
        return
    socket_path = SCREEN_DIR / f"{console['pid']}.{console['session']}"
    try:
        socket_path.unlink()
    except FileNotFoundError:
        pass
    except OSError:
        # The final existence check in delete_console reports a friendly error
        # if even the operating system cannot remove the stale entry.
        pass


def ensure_screenrc():
    if not SCREENRC.exists():
        SCREENRC.write_text(DEFAULT_SCREENRC, encoding="utf-8")


def create_console(name):
    session = session_name(name)
    if console_exists(name):
        raise ApiError("That console already exists.", 409)
    ensure_screenrc()
    bash = shutil.which("bash") or "/data/data/com.termux/files/usr/bin/bash"
    history_file = HISTORY_DIR / (name + ".bash_history")
    env = SCREEN_ENV.copy()
    env["HISTFILE"] = str(history_file)
    env["HISTSIZE"] = "10000"
    env["HISTFILESIZE"] = "20000"
    env["AC_CONSOLE_NAME"] = name
    result = subprocess.run(
        ["screen", "-c", str(SCREENRC), "-dmS", session, bash],
        env=env,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
        timeout=6,
    )
    if result.returncode != 0:
        raise ApiError("screen could not create the console; check the Web UI log.", 500)
    for _ in range(20):
        if console_exists(name):
            return
        time.sleep(0.1)
    raise ApiError("The console was started but did not appear in screen -ls.", 500)


def delete_console(name):
    session = session_name(name)
    if find_console(name) is None:
        return  # idempotent: deleting an already-gone console is still success

    # A live/attached console closes here. Remote or dead sessions cannot
    # receive -X quit, so never show screen's raw error: continue to wipe them.
    try:
        run_screen(["-S", session, "-X", "quit"])
    except (OSError, subprocess.TimeoutExpired):
        pass
    time.sleep(0.15)

    # screen -wipe [match] removes destroyed/unreachable sessions. Target only
    # this console; do NOT wipe unrelated dead screens belonging to the user.
    if find_console(name) is not None:
        try:
            run_screen(["-wipe", session])
        except (OSError, subprocess.TimeoutExpired):
            pass
        time.sleep(0.1)

    # Some Termux/screen builds leave a stale socket even after targeted wipe.
    # Remove it only while screen -ls still explicitly marks THIS entry dead.
    current = find_console(name)
    if current is not None and current.get("dead"):
        remove_dead_socket(current)

    # Retry with the unique pid.session matcher used by screen itself.
    current = find_console(name)
    if current is not None and current.get("dead"):
        try:
            run_screen(["-wipe", f"{current['pid']}.{session}"])
        except (OSError, subprocess.TimeoutExpired):
            pass
        current = find_console(name)
        if current is not None and current.get("dead"):
            remove_dead_socket(current)

    if find_console(name) is not None:
        raise ApiError("This console is still active and could not be closed. Open it, run 'exit', then try Delete again.", 500)


def clean_hardcopy(raw):
    text = ANSI_RE.sub("", raw).replace("\x00", "").replace("\r\n", "\n").replace("\r", "\n")
    lines = [line.rstrip() for line in text.split("\n")]
    while lines and not lines[-1]:
        lines.pop()
    text = "\n".join(lines)
    truncated = len(text) > MAX_OUTPUT_CHARS
    if truncated:
        text = "… older output truncated by the Web UI …\n" + text[-MAX_OUTPUT_CHARS:]
    return text, truncated


def read_output(name):
    session = session_name(name)
    fd, path_string = tempfile.mkstemp(prefix=name + "-", suffix=".hardcopy", dir=str(OUTPUT_DIR))
    os.close(fd)
    path = Path(path_string)
    try:
        # screen expects to create the hardcopy itself.
        path.unlink()
        result = run_screen(["-S", session, "-X", "hardcopy", "-h", str(path)])
        if result.returncode != 0 or not path.exists():
            detail = (result.stderr or result.stdout).strip()
            raise ApiError("Console not found or its output is unavailable." + (" " + detail if detail else ""), 404)
        raw = path.read_text(encoding="utf-8", errors="replace")
        text, truncated = clean_hardcopy(raw)
        return {"output": text, "truncated": truncated}
    finally:
        try:
            path.unlink()
        except FileNotFoundError:
            pass


def remember_command(name, command):
    # Commands entered through the Web UI also become completion suggestions.
    # The full output/history still comes from the real screen scrollback.
    path = HISTORY_DIR / (name + ".web_history")
    with HISTORY_LOCK:
        with path.open("a", encoding="utf-8") as handle:
            handle.write(command.replace("\n", " ") + "\n")
        try:
            lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
            if len(lines) > 300:
                path.write_text("\n".join(lines[-200:]) + "\n", encoding="utf-8")
        except OSError:
            pass


def execute_command(name, command):
    session = session_name(name)
    if not isinstance(command, str) or not command.strip():
        raise ApiError("Enter a command first.")
    if len(command) > MAX_COMMAND_CHARS:
        raise ApiError("Command is too long (4000 characters maximum).")
    if "\x00" in command or "\n" in command or "\r" in command:
        raise ApiError("Enter one command line at a time.")
    if not console_exists(name):
        raise ApiError("Console not found.", 404)
    # No shell=True here: the command is pasted into the existing interactive
    # shell, exactly as if it was typed in Termux. This also preserves Termux's
    # own command-not-found / "did you mean" output.
    result = run_screen(["-S", session, "-X", "stuff", command + "\n"])
    if result.returncode != 0:
        raise ApiError("Could not send the command: " + (result.stderr.strip() or "screen error"), 500)
    remember_command(name, command)


def recent_web_history(name):
    path = HISTORY_DIR / (name + ".web_history")
    try:
        lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    except OSError:
        return []
    return list(reversed(lines[-150:]))


def available_commands():
    now = time.monotonic()
    with COMMAND_CACHE_LOCK:
        if now - COMMAND_CACHE["at"] < 60 and COMMAND_CACHE["commands"]:
            return COMMAND_CACHE["commands"]
    commands = {
        "alias", "bg", "break", "cd", "clear", "continue", "declare", "dirs",
        "disown", "echo", "eval", "exec", "exit", "export", "fg", "help",
        "history", "jobs", "kill", "popd", "printf", "pushd", "pwd", "read",
        "readonly", "set", "shift", "source", "test", "times", "trap", "type",
        "typeset", "ulimit", "umask", "unalias", "unset", "wait",
    }
    for folder in os.environ.get("PATH", "").split(os.pathsep):
        if not folder:
            continue
        try:
            with os.scandir(folder) as entries:
                for entry in entries:
                    try:
                        if entry.is_file() and os.access(entry.path, os.X_OK):
                            commands.add(entry.name)
                    except OSError:
                        continue
        except OSError:
            continue
    result = sorted(commands)
    with COMMAND_CACHE_LOCK:
        COMMAND_CACHE["at"] = now
        COMMAND_CACHE["commands"] = result
    return result


def suggest_commands(name, query):
    validate_name(name)
    if not query or len(query) > 80:
        return []
    lowered = query.lower()
    suggestions = []
    seen = set()
    for command in recent_web_history(name) + available_commands():
        if command.lower().startswith(lowered) and command not in seen:
            suggestions.append(command)
            seen.add(command)
        if len(suggestions) >= 12:
            break
    return suggestions


def get_stats():
    memory = {"used": 0, "total": 0, "percent": 0}
    try:
        values = {}
        with open("/proc/meminfo", encoding="utf-8") as handle:
            for line in handle:
                parts = line.split()
                if len(parts) >= 2 and parts[0].endswith(":"):
                    values[parts[0][:-1]] = int(parts[1])
        total = values.get("MemTotal", 0) // 1024
        available = values.get("MemAvailable")
        if available is None:
            available = values.get("MemFree", 0) + values.get("Buffers", 0) + values.get("Cached", 0)
        free = available // 1024
        used = max(0, total - free)
        memory = {"used": used, "total": total, "percent": round(used * 100 / total) if total else 0}
    except (OSError, ValueError):
        pass
    try:
        with open("/proc/loadavg", encoding="utf-8") as handle:
            load = " ".join(handle.read().split()[:3])
    except OSError:
        load = "N/A"
    disk = {"used": "", "total": "", "percent": ""}
    try:
        usage = shutil.disk_usage(str(Path.home()))
        disk = {
            "used": format_bytes(usage.used),
            "total": format_bytes(usage.total),
            "percent": str(round(usage.used * 100 / usage.total)) + "%" if usage.total else "0%",
        }
    except OSError:
        pass
    return {"load": load, "memory": memory, "disk": disk}


def format_bytes(value):
    value = float(value)
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if value < 1024 or unit == "TB":
            return f"{value:.1f} {unit}" if unit != "B" else f"{int(value)} B"
        value /= 1024


class Handler(http.server.BaseHTTPRequestHandler):
    server_version = "AurerConsole/2.0"
    sys_version = ""

    # Never write HTTP access logs to the terminal running the ac dashboard.
    def log_message(self, *args):
        return

    def common_headers(self):
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("Referrer-Policy", "no-referrer")
        self.send_header("Cache-Control", "no-store")
        self.send_header("Content-Security-Policy", "default-src 'self'; script-src 'unsafe-inline'; style-src 'unsafe-inline'; connect-src 'self'; img-src 'self' data:; base-uri 'none'; frame-ancestors 'none'")

    def send_json(self, data, code=200):
        body = json.dumps(data, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.common_headers()
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def send_html(self):
        body = HTML.encode("utf-8")
        self.send_response(200)
        self.common_headers()
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def read_body(self):
        try:
            length = int(self.headers.get("Content-Length", "0"))
        except ValueError:
            raise ApiError("Invalid Content-Length.")
        if length < 0 or length > MAX_BODY_BYTES:
            raise ApiError("Request body is too large.", 413)
        if not length:
            return {}
        try:
            data = json.loads(self.rfile.read(length))
        except (json.JSONDecodeError, UnicodeDecodeError):
            raise ApiError("Invalid JSON request.")
        if not isinstance(data, dict):
            raise ApiError("JSON body must be an object.")
        return data

    def handle_exception(self, error):
        if isinstance(error, ApiError):
            self.send_json({"ok": False, "error": error.message}, error.status)
        elif isinstance(error, subprocess.TimeoutExpired):
            self.send_json({"ok": False, "error": "A screen operation timed out. Try again."}, 504)
        elif isinstance(error, (BrokenPipeError, ConnectionResetError)):
            return
        else:
            # This goes to webui.log because webui_start redirects both stdout
            # and stderr there. It can never draw over the interactive dashboard.
            traceback.print_exc()
            self.send_json({"ok": False, "error": "Internal Web UI error; see ~/.ac-consoles/webui.log."}, 500)

    def do_GET(self):
        try:
            parsed = urllib.parse.urlsplit(self.path)
            query = urllib.parse.parse_qs(parsed.query)
            if parsed.path in ("/", "/index.html"):
                self.send_html()
            elif parsed.path == "/api/state":
                self.send_json({"ok": True, "consoles": list_consoles(), "stats": get_stats()})
            elif parsed.path == "/api/output":
                name = validate_name(query.get("name", [""])[0])
                data = read_output(name)
                self.send_json({"ok": True, "name": name, **data})
            elif parsed.path == "/api/suggest":
                name = query.get("name", [""])[0]
                q = query.get("q", [""])[0]
                self.send_json({"ok": True, "suggestions": suggest_commands(name, q)})
            else:
                self.send_json({"ok": False, "error": "Not found."}, 404)
        except Exception as error:
            self.handle_exception(error)

    def do_POST(self):
        try:
            path = urllib.parse.urlsplit(self.path).path
            body = self.read_body()
            if path == "/api/create":
                name = validate_name(body.get("name", ""))
                create_console(name)
                self.send_json({"ok": True, "name": name})
            elif path == "/api/delete":
                name = validate_name(body.get("name", ""))
                delete_console(name)
                self.send_json({"ok": True})
            elif path == "/api/execute":
                name = validate_name(body.get("name", ""))
                execute_command(name, body.get("command", ""))
                self.send_json({"ok": True})
            else:
                self.send_json({"ok": False, "error": "Not found."}, 404)
        except Exception as error:
            self.handle_exception(error)

delete_console() {
    local name="$1" pid=""
    [ -n "$name" ] || { echo "Usage: ac delete <name>"; return 1; }
    if ! exists "$name"; then
        echo "Console '$name' is already gone."
        return 0
    fi

    # Live consoles close with quit. Remote/dead consoles reject it, so hide
    # that expected screen error and continue through the stale-session cleanup.
    screen -S "${SESSION_PREFIX}${name}" -X quit >/dev/null 2>&1 || true
    sleep 0.15

    # Target only this session; never wipe unrelated screens.
    if exists "$name"; then
        screen -wipe "${SESSION_PREFIX}${name}" >/dev/null 2>&1 || true
        sleep 0.1
    fi

    # Fallback for Termux/screen builds that leave the dead socket behind.
    if exists "$name" && console_is_dead "$name"; then
        remove_dead_socket "$name"
    fi

    # Retry using screen's unique pid.session matcher, then clean the exact
    # socket again if it is still explicitly marked Remote or dead.
    if exists "$name" && console_is_dead "$name"; then
        pid="$(console_pid "$name")"
        if [ -n "$pid" ]; then
            screen -wipe "$pid.${SESSION_PREFIX}${name}" >/dev/null 2>&1 || true
        fi
        if exists "$name" && console_is_dead "$name"; then
            remove_dead_socket "$name"
        fi
    fi

    if exists "$name"; then
        echo "ERROR: Console '$name' is still active. Open it, run 'exit', then delete it again."
        return 1
    fi
    echo "Console '$name' deleted."
}

    load=$(awk '{print $1" "$2" "$3}' /proc/loadavg 2>/dev/null || echo "N/A")
    web=$(webui_status)
    echo "Load:$load RAM:${mem_used:-?}/${mem_total:-?}MB($mem_pct%) $web"
}

show_help() {
cat <<EOF
Aurer Console (ac):
  ac                 Dashboard
  ac list            List consoles
  ac create <name>   Create console
  ac open <name>     Open console in Termux
  ac delete <name>   Delete console
  ac webui           Toggle Web UI
  ac webui-start     Start Web UI at http://localhost:$WEB_PORT
  ac webui-stop      Stop Web UI
  ac webui-status    Show Web UI status

Web UI:
  Open a console to see its real screen output and run commands in that
  same shell. Existing output is restored from screen scrollback.

Exit a console in Termux:
  Double-tap q q quickly.
EOF
}

HEADER_ROWS=5
STATS_ROW=4

draw_full() {
    local names stats
    names=$(list_names)
    stats=$(get_stats)
    ansi_clear
    printf "╔════════════════════════════════════════╗\n"
    printf "║  %-38s║\n" "Aurer Console Manager"
    printf "╠════════════════════════════════════════╣\n"
    printf "║  %-38s║\n" "$stats"
    printf "╠════════════════════════════════════════╣\n"
    if [ -z "$names" ]; then
        printf "║  %-38s║\n" "(no consoles running)"
    else
        while IFS= read -r n; do printf "║  ▶ %-36s║\n" "$n"; done <<< "$names"
    fi
    printf "╠════════════════════════════════════════╣\n"
    printf "║  %-38s║\n" "[1]Open [2]Create [3]Delete [4]Help"
    printf "║  %-38s║\n" "[n]Toggle WebUI  [q]Quit"
    printf "╚════════════════════════════════════════╝\n"
    printf "> "
}

update_stats_inline() {
    local stats
    stats=$(get_stats)
    ansi_save
    ansi_goto "$STATS_ROW"
    printf "║  %-38s║" "$stats"
    ansi_restore
}

dashboard() {
    local prev_names="" prev_stats="" curr_names curr_stats drawn=0 choice n
    while true; do
        curr_names=$(list_names | sort | tr '\n' '|')
        curr_stats=$(get_stats)
        if [ "$curr_names" != "$prev_names" ] || [ "$drawn" -eq 0 ]; then
            draw_full
            prev_names="$curr_names"
            prev_stats="$curr_stats"
            drawn=1
        elif [ "$curr_stats" != "$prev_stats" ]; then
            update_stats_inline
            prev_stats="$curr_stats"
        fi
        if read -t 1 -rp "" choice 2>/dev/null; then
            printf "\n"
            case "$choice" in
                1) read -rp "Name to open: " n; open_console "$n"; drawn=0 ;;
                2) read -rp "New name: " n; create_console "$n"; sleep 0.4; drawn=0 ;;
                3) read -rp "Name to delete: " n; delete_console "$n"; sleep 0.4; drawn=0 ;;
                4) show_help; read -rp "Press enter..."; drawn=0 ;;
                n|N) webui_toggle; sleep 0.6; drawn=0 ;;
                q|Q|0) ansi_clear; exit 0 ;;
                "") drawn=0 ;;
                *) echo "Invalid"; sleep 0.4; drawn=0 ;;
            esac
        fi
    done
}

case "${1:-}" in
    list) list_names ;;
    create) create_console "${2:-}" ;;
    open) open_console "${2:-}" ;;
    delete) delete_console "${2:-}" ;;
    webui) webui_toggle ;;
    webui-start) webui_start ;;
    webui-stop) webui_stop ;;
    webui-status) webui_status; echo ;;
    help|--help|-h) show_help ;;
    "") dashboard ;;
    *) echo "Unknown command."; show_help; exit 1 ;;
esac
